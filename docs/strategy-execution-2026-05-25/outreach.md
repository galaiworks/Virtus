### スコアリングルーブリック

```python
def score_summit_attendee(attendee: dict) -> int:
    score = 0

    # 業界適合 (max 30)
    if attendee.industry in TARGET_INDUSTRIES:
        score += 30
    elif attendee.industry in ADJACENT_INDUSTRIES:
        score += 15

    # 決済権 (max 25)
    if attendee.role in ["代表取締役", "CEO", "創業者", "経営者"]:
        score += 25
    elif attendee.role in ["COO", "事業責任者"]:
        score += 15

    # 痛みあり (max 25)
    pain_signals = [
        "営業効率",
        "コンテンツ発信",
        "属人化",
        "時間がない",
        "ひとり",
        "DX",
    ]
    if any(signal in attendee.申込動機 for signal in pain_signals):
        score += 25

    # 儲かってそう (max 20)
    if attendee.company_size in ["1-10名", "11-50名"]:
        if has_recent_funding_or_growth_signal(attendee):
            score += 20
        else:
            score += 10

    return min(score, 100)
```

### 出力スキーマ

```yaml
summit_attendee:

  role: str
  industry: str
  score: int  # 0-100
```
